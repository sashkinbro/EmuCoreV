cmake_minimum_required(VERSION 3.22)
project(EmuCoreVNativeUi LANGUAGES CXX)

# Standalone reconstruction: no guest CPU, emulator, SDL or game dependencies.
add_library(emucorev_paf STATIC paf/src/resource_document.cpp paf/src/animation_curve.cpp paf/src/animation_timer.cpp paf/src/gim_image.cpp paf/src/bubble_shaders.cpp paf/src/bubble_motion.cpp paf/src/busy_indicator.cpp)
target_include_directories(emucorev_paf PUBLIC paf/include)
target_compile_features(emucorev_paf PUBLIC cxx_std_20)
if(ANDROID)
    target_link_libraries(emucorev_paf PUBLIC z)
else()
    find_package(ZLIB REQUIRED)
    target_link_libraries(emucorev_paf PUBLIC ZLIB::ZLIB)
endif()
if(MSVC)
    target_compile_options(emucorev_paf PRIVATE /W4 /WX /fp:strict)
else()
    target_compile_options(emucorev_paf PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endif()

if(ANDROID)
    add_library(emucorev_paf_gles STATIC paf/src/bubble_renderer.cpp paf/src/sprite_renderer.cpp)
    target_include_directories(emucorev_paf_gles PUBLIC paf/include)
    target_compile_features(emucorev_paf_gles PUBLIC cxx_std_20)
    target_compile_options(emucorev_paf_gles PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(emucorev_paf_gles PUBLIC GLESv3)
    add_executable(paf_bubble_preview tools/bubble_preview.cpp)
    target_link_libraries(paf_bubble_preview PRIVATE emucorev_paf_gles emucorev_paf EGL)
endif()

add_executable(paf_resource_inspect tools/resource_inspect.cpp)
target_link_libraries(paf_resource_inspect PRIVATE emucorev_paf)

add_executable(paf_gim_decode tools/gim_decode.cpp)
target_link_libraries(paf_gim_decode PRIVATE emucorev_paf)
add_executable(paf_export_images tools/export_images.cpp)
target_link_libraries(paf_export_images PRIVATE emucorev_paf)

include(CTest)
if(BUILD_TESTING)
    add_executable(paf_busy_test tests/busy_indicator_test.cpp)
    target_link_libraries(paf_busy_test PRIVATE emucorev_paf)
    add_executable(paf_bubble_motion_test tests/bubble_motion_test.cpp)
    target_link_libraries(paf_bubble_motion_test PRIVATE emucorev_paf)
    add_executable(paf_resource_test tests/resource_document_test.cpp)
    target_link_libraries(paf_resource_test PRIVATE emucorev_paf)
    add_test(NAME paf_resource_document COMMAND paf_resource_test)
    add_executable(paf_curve_test tests/animation_curve_test.cpp)
    target_link_libraries(paf_curve_test PRIVATE emucorev_paf)
    add_test(NAME paf_animation_curve COMMAND paf_curve_test)
    add_executable(paf_timer_test tests/animation_timer_test.cpp)
    target_link_libraries(paf_timer_test PRIVATE emucorev_paf)
    add_executable(paf_gim_test tests/gim_image_test.cpp)
    target_link_libraries(paf_gim_test PRIVATE emucorev_paf)
    add_test(NAME paf_gim_image COMMAND paf_gim_test)
    if(ANDROID)
        # Research-only GPU compiler test. Requires externally supplied GLSL;
        # never linked into emucorev_paf or the Android application.
        add_executable(paf_shader_compile_test tests/shader_compile_device.cpp)
        target_compile_features(paf_shader_compile_test PRIVATE cxx_std_20)
        target_link_libraries(paf_shader_compile_test PRIVATE EGL GLESv3)
        add_executable(paf_shader_pipeline_test tests/shader_pipeline_device.cpp)
        target_compile_features(paf_shader_pipeline_test PRIVATE cxx_std_20)
        target_link_libraries(paf_shader_pipeline_test PRIVATE vulkan)
    endif()
endif()
