cmake_minimum_required(VERSION 3.22)
project(EmuCoreVNativeUi LANGUAGES CXX)

# Standalone reconstruction: no guest CPU, emulator, SDL or game dependencies.
add_library(emucorev_paf STATIC paf/src/resource_document.cpp paf/src/animation_curve.cpp paf/src/animation_timer.cpp paf/src/gim_image.cpp paf/src/bubble_shaders.cpp paf/src/bubble_motion.cpp paf/src/busy_indicator.cpp paf/src/page_turn.cpp paf/src/page_turn_material.cpp paf/src/page_turn_shaders.cpp paf/src/plane_div.cpp paf/src/wave_preset.cpp paf/src/wave_simulation.cpp paf/src/wave_scene.cpp)
target_include_directories(emucorev_paf PUBLIC paf/include)
target_compile_features(emucorev_paf PUBLIC cxx_std_20)
if(ANDROID)
    target_link_libraries(emucorev_paf PUBLIC z)
else()
    find_package(ZLIB REQUIRED)
    target_link_libraries(emucorev_paf PUBLIC ZLIB::ZLIB)
endif()
if(MSVC)
    target_compile_options(emucorev_paf PRIVATE /W4 /WX /fp:strict)
else()
    target_compile_options(emucorev_paf PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endif()

if(ANDROID)
    add_library(emucorev_paf_gles STATIC paf/src/bubble_renderer.cpp paf/src/sprite_renderer.cpp paf/src/page_turn_renderer.cpp paf/src/wave_position_renderer.cpp paf/src/wave_surface_renderer.cpp paf/src/wave_grid_normal_renderer.cpp paf/src/wave_lighting_renderer.cpp paf/src/wave_skydome_renderer.cpp paf/src/wave_renderer.cpp)
    target_include_directories(emucorev_paf_gles PUBLIC paf/include)
    target_compile_features(emucorev_paf_gles PUBLIC cxx_std_20)
    target_compile_options(emucorev_paf_gles PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(emucorev_paf_gles PUBLIC GLESv3 emucorev_paf)
    add_executable(paf_bubble_preview tools/bubble_preview.cpp)
    target_link_libraries(paf_bubble_preview PRIVATE emucorev_paf_gles emucorev_paf EGL)
endif()

add_executable(paf_resource_inspect tools/resource_inspect.cpp)
